CREATE TABLE DIARY_TASKS ( TASK_ID NUMBER PRIMARY KEY,USER_ID NUMBER NOT NULL,TASK_TITLE VARCHAR2(200) NOT NULL,TASK_DESCRIPTION VARCHAR2(1000),TASK_DATE DATE DEFAULT SYSDATE,DUE_DATE DATE,STATUS VARCHAR2(20) DEFAULT 'PENDING',CONSTRAINT FK_TASK_USER FOREIGN KEY (USER_ID) REFERENCES USERS(USER_ID));

CREATE TABLE DIARY_ATTACHMENTS (ATTACHMENT_ID NUMBER PRIMARY KEY, NOTE_ID NUMBER NOT NULL,FILE_NAME VARCHAR2(255),FILE_PATH VARCHAR2(1000),ATTACHMENT_DATE DATE DEFAULT SYSDATE, CONSTRAINT FK_ATTACHMENT_NOTE FOREIGN KEY (NOTE_ID) REFERENCES DIARY_NOTES(NOTE_ID));

CREATE TABLE DAILY_GOALS ( GOAL_ID NUMBER PRIMARY KEY,USER_ID NUMBER NOT NULL,GOAL_TITLE VARCHAR2(200) NOT NULL,GOAL_DATE DATE DEFAULT SYSDATE,STATUS VARCHAR2(20) DEFAULT 'PENDING', CONSTRAINT FK_GOAL_USER FOREIGN KEY (USER_ID) REFERENCES USERS(USER_ID));
